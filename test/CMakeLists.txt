# Big full-transverse-mesh 3D test — needs substantial RAM as well as time, so
# it is gated separately from PUFFIN_SLOW_TESTS and carries both labels.
#
#   cmake -B build -DPUFFIN_BIG_TESTS=ON .
#   cmake --build build
#   ctest --test-dir build -L big
#
# The default of 6 ranks is the performance-core count of the development
# machine (an Apple M5 Pro: 6 P-cores + 12 E-cores). Letting it spill onto the
# E-cores does buy about 40% — 16 ranks finishes the integration in 5.4 min
# against 7.5 min for 6 — but 6 keeps the test on cores of one kind, so its
# timings mean something and do not depend on how macOS happens to schedule
# across core types.
#
# Measured over two full runs at 6 ranks: 522 s and 532 s wall clock, peaking at
# 1.4-2.0 GB resident per rank (9-12 GB in total) and writing ~1.3 GB.
#
# Memory per rank scales roughly as 1/nranks, so lowering the rank count raises
# the per-rank footprint. It also grows with the number of steps taken -- about
# 750 MB per rank at 300 steps against 1.4 GB at 7500 -- so size the machine
# from the full run, not from a quick one.
#
# PUFFIN_BIG_TEST_PES must match the npes= in testMPIIntegration3DBig.pf, and
# the machine must actually have that many slots or mpiexec will refuse to
# launch. Changing it also invalidates the golden reference: the test pins the
# beam's shot noise with iRandSeed, but each rank seeds its own stream, so the
# realisation depends on the decomposition. See testMPIIntegration3DBig.pf.
option(PUFFIN_BIG_TESTS "Build big integration tests (85x85x5780 field mesh, ~9-12GB RAM and ~9 min at 6 ranks)" OFF)
set(PUFFIN_BIG_TEST_PES 6 CACHE STRING "MPI ranks used by the big 3D integration test")

if(PUFFIN_BIG_TESTS)

  configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/inputs/3D/shcse_big.in"
    "${CMAKE_CURRENT_BINARY_DIR}/inputs/3D/shcse_big.in"
    COPYONLY
  )

  configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/inputs/3D/shcse_big_beam.in"
    "${CMAKE_CURRENT_BINARY_DIR}/inputs/3D/shcse_big_beam.in"
    COPYONLY
  )

  # Only the integrated file is small enough to keep under version control
  # (~400 KB). The field and electron dumps are 637 MB and 20 MB, so the test
  # checks those against reductions hard-coded in the .pf instead.
  configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/expected/3D/shcse_big_expected_integrated.h5"
    "${CMAKE_CURRENT_BINARY_DIR}/expected/3D/shcse_big_expected_integrated.h5"
    COPYONLY
  )

  # The rank count lives in two places -- MAX_PES here, and npes= in the .pf --
  # and they must agree. If they do not, pFUnit launches one number of ranks
  # while the test body expects another, and the run deadlocks with no output
  # and no error: it just sits there until the ctest timeout. Note that editing
  # the default above does *not* change an existing build, because `set(...
  # CACHE ...)` leaves a cached value alone, so this is easy to trip over.
  # Check it at configure time and fail loudly instead.
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/testMPIIntegration3DBig.pf" _big_pf_src)
  string(REGEX MATCH "@mpitest\\(npes=\\[([0-9]+)\\]\\)" _big_pf_match "${_big_pf_src}")
  if(NOT _big_pf_match)
    message(FATAL_ERROR
      "Could not find an @mpitest(npes=[N]) in testMPIIntegration3DBig.pf, so "
      "the rank count cannot be checked against PUFFIN_BIG_TEST_PES.")
  endif()
  if(NOT CMAKE_MATCH_1 STREQUAL "${PUFFIN_BIG_TEST_PES}")
    message(FATAL_ERROR
      "Rank count mismatch: PUFFIN_BIG_TEST_PES is ${PUFFIN_BIG_TEST_PES} but "
      "testMPIIntegration3DBig.pf declares npes=[${CMAKE_MATCH_1}]. These must "
      "agree or the test will hang. Either pass "
      "-DPUFFIN_BIG_TEST_PES=${CMAKE_MATCH_1}, or change npes= in the .pf to "
      "${PUFFIN_BIG_TEST_PES} and regenerate the golden reference (the shot "
      "noise realisation depends on the rank count).")
  endif()

  add_pfunit_ctest(puffin_e2e_tests_3d_big
    TEST_SOURCES testMPIIntegration3DBig.pf
    LINK_LIBRARIES puffin_mpi
    MAX_PES ${PUFFIN_BIG_TEST_PES}
  )

  set_tests_properties(puffin_e2e_tests_3d_big PROPERTIES
    LABELS "slow;big"
    TIMEOUT 3600
    RUN_SERIAL TRUE
  )

  # Recomputes the test's hard-coded reduction constants from output files that
  # already exist, so regenerating them does not need a second full simulation.
  # Not a test — it is a tool, so it is built but never registered with ctest.
  add_executable(harvest_big_reference harvest_big_reference.f90)
  target_link_libraries(harvest_big_reference puffin_mpi)

endif()
